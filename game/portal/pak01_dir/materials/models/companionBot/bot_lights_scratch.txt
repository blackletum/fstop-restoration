	"$var_blink_thresh" "0.65"

	"proxies"
	{
		//-------------------------------------------
		// setting the blink period from hp
		//-------------------------------------------
		"EntityHealth"
		{
			"resultVar" "$var_hp_frac"
		}

		"Clamp"
		{
			"resultVar" "$var_hp_frac"
			"srcVar1" "$var_hp_frac"
			"min" ".25"
			"max" "1"
		}
	
		"Sine"
		{
			"resultVar" "$var_blink_period"
			"sineperiod" "$var_hp_frac"
			"sinemin" "0"
			"sinemax" "1"
		}	

		//-------------------------------------------
		//set blink factor:
		//to constant 1 if hp is above our thresh (no blinking),
		//otherwise use the sine result to blink
		//-------------------------------------------
		"LessOrEqual"
		{
			"resultVar" "$var_blink_factor"
			"srcVar1" "$var_hp_frac"
			"srcVar2" "$var_blink_thresh"
			"lessEqualVar" "$var_blink_period"
			"greaterVar" "$var_1"
		}
		


		//--------------------------------------------
		//setting components of illumtint to go from green-red 
		//based on health.  try 3 stages
		//-------------------------------------------
		
		//green is the hp frac
		"EntityHealth"
		{
			"resultVar" "$var_color_g"
		}
		
		//blue component is zero all the time
		"Equals"
		{
			"srcVar1" "$var_0"
			"resultVar" "$selfillumtint[2]"
		}

		//red is 1 - hp frac
		"Subtract"
		{
			"srcVar1" "$var_1"
			"srcVar2" "$var_color_g"
			"resultVar" "$var_color_r"
		}

		//-------------------------------------------
		// multiply each tint component by the blink factor
		//-------------------------------------------
		"Multiply"
		{
			"srcVar1" "$var_color_r"
			"srcVar2" "$var_blink_factor"
			"resultVar" "$selfillumtint[0]"
		}

		"Multiply"
		{
			"srcVar1" "$var_color_g"
			"srcVar2" "$var_blink_factor"
			"resultVar" "$selfillumtint[1]"
		}
